#include <Arduino.h>
#include "esp_camera.h"
#include <WiFi.h>
#include <HTTPClient.h>
#include <WebServer.h>
#include <Preferences.h>
#include "FS.h"
#include "SD_MMC.h"
#include "board_config.h"

Preferences preferences;
WebServer setupServer(80);

String savedSSID;
String savedPassword;
String serverUrl;

void startCameraServer();
void setupLedFlash();

void saveCredentials(String ssid, String password, String serverIp) {
  preferences.begin("eosas", false);
  preferences.putString("ssid", ssid);
  preferences.putString("password", password);
  preferences.putString("serverIp", serverIp);
  preferences.end();

  Serial.println("Saved credentials.");
}

void loadCredentials() {
  preferences.begin("eosas", true);
  savedSSID = preferences.getString("ssid", "");
  savedPassword = preferences.getString("password", "");
  String serverIp = preferences.getString("serverIp", "");
  preferences.end();

  Serial.print("Loaded SSID: ");
  Serial.println(savedSSID);
  Serial.print("Loaded Server IP: ");
  Serial.println(serverIp);

  if (serverIp.length() > 0) {
    serverUrl = "http://" + serverIp + ":5000/predict";
  }
}

void startSetupPortal() {
  WiFi.mode(WIFI_AP);
  WiFi.softAP("EOSAS_Setup", "eosas1234");
  delay(1000);

  Serial.print("Setup AP IP address: ");
  Serial.println(WiFi.softAPIP());

  setupServer.on("/", HTTP_GET, []() {
    String html = "";
    html += "<html><body style='font-family:Arial;'>";
    html += "<h2>EOSAS Wi-Fi Setup</h2>";
    html += "<form action='/save' method='POST'>";
    html += "Wi-Fi Name:<br><input name='ssid'><br><br>";
    html += "Wi-Fi Password:<br><input name='password' type='password'><br><br>";
    html += "Flask Server IP:<br><input name='serverIp' placeholder='192.168.1.39'><br><br>";
    html += "<button type='submit'>Save and Restart</button>";
    html += "</form></body></html>";
    setupServer.send(200, "text/html", html);
  });

  setupServer.on("/save", HTTP_POST, []() {
    String newSSID = setupServer.arg("ssid");
    String newPassword = setupServer.arg("password");
    String newServerIp = setupServer.arg("serverIp");

    Serial.println("Save request received.");
    Serial.println(newSSID);
    Serial.println(newServerIp);

    saveCredentials(newSSID, newPassword, newServerIp);

    setupServer.send(200, "text/html",
      "<h2>Saved!</h2><p>ESP32-CAM restarting in 5 seconds.</p>");

    delay(5000);
    ESP.restart();
  });

  setupServer.begin();

  Serial.println("Setup portal running. Connect to EOSAS_Setup and open 192.168.4.1");

  while (true) {
    setupServer.handleClient();
    delay(500);
  }
}

bool connectToSavedWiFi() {
  loadCredentials();

  if (savedSSID.length() == 0 || serverUrl.length() == 0) {
    Serial.println("No saved Wi-Fi/server settings.");
    return false;
  }

  WiFi.mode(WIFI_STA);
  WiFi.begin(savedSSID.c_str(), savedPassword.c_str());
  WiFi.setSleep(false);

  Serial.print("Connecting to saved Wi-Fi");

  int attempts = 0;
  while (WiFi.status() != WL_CONNECTED && attempts < 30) {
    delay(500);
    Serial.print(".");
    attempts++;
  }

  Serial.println();

  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("Wi-Fi connected!");
    Serial.print("ESP32 IP: ");
    Serial.println(WiFi.localIP());
    Serial.print("Server URL: ");
    Serial.println(serverUrl);
    return true;
  }

  Serial.println("Wi-Fi failed.");
  return false;
}

void savePhotoToSD(camera_fb_t *fb) {

  if (!fb) {
    Serial.println("No frame buffer");
    return;
  }

  String path = "/scan_" + String(millis()) + ".jpg";

  File file = SD_MMC.open(path.c_str(), FILE_WRITE);

  if (!file) {
    Serial.println("Failed to open file for writing");
    return;
  }

  file.write(fb->buf, fb->len);

  file.close();

  Serial.print("Saved file to SD: ");
  Serial.println(path);
}

bool shouldScan() {

  WiFiClient client;
  HTTPClient http;

  String commandUrl = serverUrl;
  commandUrl.replace("/predict", "/get_command");

  http.begin(client, commandUrl);

  int code = http.GET();

  if (code == 200) {

    String response = http.getString();

    http.end();

    if (response.indexOf("\"scan\":true") >= 0) {
      return true;
    }
  }

  http.end();

  return false;
}

void sendPhotoToServer() {
  camera_fb_t * fb = esp_camera_fb_get();

  if (!fb) {
    Serial.println("Camera capture failed");
    return;
  }

savePhotoToSD(fb);

  WiFiClient client;
  HTTPClient http;

  http.begin(client, serverUrl);
  http.setTimeout(15000);

  String boundary = "----EOSASBoundary";
  String contentType = "multipart/form-data; boundary=" + boundary;
  http.addHeader("Content-Type", contentType);

  String bodyStart = "--" + boundary + "\r\n";
  bodyStart += "Content-Disposition: form-data; name=\"image\"; filename=\"capture.jpg\"\r\n";
  bodyStart += "Content-Type: image/jpeg\r\n\r\n";

  String bodyEnd = "\r\n--" + boundary + "--\r\n";

  int totalLen = bodyStart.length() + fb->len + bodyEnd.length();

  uint8_t *postData = (uint8_t *)malloc(totalLen);

  if (!postData) {
    Serial.println("Not enough memory for POST data");
    esp_camera_fb_return(fb);
    return;
  }

  memcpy(postData, bodyStart.c_str(), bodyStart.length());
  memcpy(postData + bodyStart.length(), fb->buf, fb->len);
  memcpy(postData + bodyStart.length() + fb->len, bodyEnd.c_str(), bodyEnd.length());

  int httpResponseCode = http.POST(postData, totalLen);

  Serial.print("HTTP Response code: ");
  Serial.println(httpResponseCode);

  String response = http.getString();
  Serial.println("Server response:");
  Serial.println(response);

  free(postData);
  esp_camera_fb_return(fb);
  http.end();
}

void setup() {
  Serial.begin(115200);
  delay(3000);
  Serial.setDebugOutput(true);
  Serial.println();
  Serial.println("EOSAS BOOTING...");

  if (!connectToSavedWiFi()) {
    startSetupPortal();
  }

  camera_config_t config;
  config.ledc_channel = LEDC_CHANNEL_0;
  config.ledc_timer = LEDC_TIMER_0;
  config.pin_d0 = Y2_GPIO_NUM;
  config.pin_d1 = Y3_GPIO_NUM;
  config.pin_d2 = Y4_GPIO_NUM;
  config.pin_d3 = Y5_GPIO_NUM;
  config.pin_d4 = Y6_GPIO_NUM;
  config.pin_d5 = Y7_GPIO_NUM;
  config.pin_d6 = Y8_GPIO_NUM;
  config.pin_d7 = Y9_GPIO_NUM;
  config.pin_xclk = XCLK_GPIO_NUM;
  config.pin_pclk = PCLK_GPIO_NUM;
  config.pin_vsync = VSYNC_GPIO_NUM;
  config.pin_href = HREF_GPIO_NUM;
  config.pin_sccb_sda = SIOD_GPIO_NUM;
  config.pin_sccb_scl = SIOC_GPIO_NUM;
  config.pin_pwdn = PWDN_GPIO_NUM;
  config.pin_reset = RESET_GPIO_NUM;
  config.xclk_freq_hz = 20000000;
  config.pixel_format = PIXFORMAT_JPEG;

  config.frame_size = FRAMESIZE_QVGA;
  config.jpeg_quality = 15;
  config.fb_count = 1;
  config.grab_mode = CAMERA_GRAB_LATEST;
  config.fb_location = CAMERA_FB_IN_DRAM;

  esp_err_t err = esp_camera_init(&config);
  if (err != ESP_OK) {
    Serial.printf("Camera init failed with error 0x%x", err);
    return;
  }

if (!SD_MMC.begin("/sdcard", true)) {
  Serial.println("SD Card Mount Failed");
} else {
  Serial.println("SD Card mounted successfully");
}

  sensor_t *s = esp_camera_sensor_get();

  if (s->id.PID == OV3660_PID) {
    s->set_vflip(s, 1);
    s->set_brightness(s, 1);
    s->set_saturation(s, -2);
  }

  s->set_framesize(s, FRAMESIZE_QVGA);

#if defined(LED_GPIO_NUM)
  setupLedFlash();
#endif

  startCameraServer();

  Serial.print("Camera Ready! Use 'http://");
  Serial.print(WiFi.localIP());
  Serial.println("' to connect");
}

void loop() {

  if (WiFi.status() == WL_CONNECTED &&
      serverUrl.length() > 0) {

      if (shouldScan()) {

          Serial.println("Scan command received.");

          sendPhotoToServer();
      }
  }

  delay(1000);
}
